# ai-lab Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Portfólio de IA 100% local: prompts estruturados, benchmark de LLMs, 3 automações com horas economizadas medidas e dashboard Streamlit com "pergunte aos dados".

**Architecture:** `ailab/llm.py` isola o Ollama. `structured.py` valida saída JSON com Pydantic (retry). Automações e benchmark gravam em SQLite; o dashboard lê o SQLite e uma camada `askdata` gera SQL somente-leitura via LLM.

**Tech Stack:** Python 3.14, Ollama (`qwen2.5:3b`, `llama3.2:3b`), requests, pydantic, pyyaml, pandas, streamlit, pytest.

**Spec:** `docs/superpowers/specs/2026-10-30-ai-lab-design.md`

## Global Constraints

- Sem API key; todo LLM via Ollama em `http://localhost:11434`.
- Modelo padrão `qwen2.5:3b`; comparação com `llama3.2:3b`.
- Dados 100% sintéticos (`data/synthetic/`), nenhum dado pessoal real.
- SQL gerado: somente `SELECT`, instrução única, conexão `mode=ro`.
- Textos de prompts e UI em português.
- Testes mockam o LLM; só o smoke test fala com o Ollama.

```python
import requests

OLLAMA_URL = "http://localhost:11434"
DEFAULT_MODEL = "qwen2.5:3b"


class LLMError(RuntimeError):
    pass


def generate(prompt, model=DEFAULT_MODEL, system=None, json_mode=False,
             temperature=0.0, timeout=300):
    payload = {"model": model, "prompt": prompt, "stream": False,
               "options": {"temperature": temperature}}
    if system:
        payload["system"] = system
    if json_mode:
        payload["format"] = "json"
    try:
        r = requests.post(f"{OLLAMA_URL}/api/generate", json=payload, timeout=timeout)
        r.raise_for_status()
        return r.json()["response"]
    except (requests.RequestException, KeyError) as e:
        raise LLMError(str(e)) from e
```

`ailab/prompts.py`:
```python
from pathlib import Path
import yaml

TEMPLATES_DIR = Path(__file__).parent / "templates"
VARIANTS = ("zero_shot", "few_shot", "cot")


def load_template(name):
    return yaml.safe_load((TEMPLATES_DIR / f"{name}.yaml").read_text(encoding="utf-8"))


def render(tpl, text, variant="few_shot"):
    if variant not in VARIANTS:
        raise ValueError(f"variante desconhecida: {variant}")
    parts = [tpl["role"], tpl["instructions"],
             "Formato de saída (JSON): " + tpl["output_format"]]
    if variant == "few_shot":
        for i, ex in enumerate(tpl["examples"], 1):
            parts.append(f"Exemplo {i}\nEntrada: {ex['input']}\nSaída: {ex['output']}")
    if variant == "cot":
        parts.append('Inclua no JSON um campo "reasoning" com um raciocínio curto antes de decidir.')
    parts.append(f"Entrada: {text}\nSaída:")
    return "\n\n".join(parts)
```
> Nota: `"Exemplo"` não aparece em zero_shot; o `output_format` dos templates não deve conter a palavra "Exemplo".

`ailab/structured.py`:
```python
import time
from dataclasses import dataclass
from pydantic import BaseModel, ValidationError
from ailab.llm import generate, DEFAULT_MODEL


@dataclass
class StructuredResult:
    obj: BaseModel | None
    ok: bool
    attempts: int
    seconds: float
    error: str | None = None


def run_structured(prompt, schema, model=DEFAULT_MODEL, retries=2, llm=generate):
    start, current, error = time.perf_counter(), prompt, None
    for attempt in range(1, retries + 2):
        raw = llm(current, model=model, json_mode=True)
        try:
            obj = schema.model_validate_json(raw)
            return StructuredResult(obj, True, attempt, time.perf_counter() - start)
        except ValidationError as e:
            error = str(e)
            current = (f"{prompt}\n\nSua resposta anterior foi inválida:\n{raw}\n"
                       f"Erro: {error}\nCorrija e responda apenas com JSON válido.")
    return StructuredResult(None, False, retries + 1, time.perf_counter() - start, error)
```

`ailab/sqlguard.py`:
```python
import re

FORBIDDEN = re.compile(
    r"\b(insert|update|delete|drop|alter|create|replace|attach|detach|pragma|vacuum)\b", re.I)


def validate_select(sql):
    cleaned = sql.strip().rstrip(";").strip()
    if ";" in cleaned:
        raise ValueError("apenas uma instrução é permitida")
    if not re.match(r"^(select|with)\b", cleaned, re.I):
        raise ValueError("apenas SELECT é permitido")
    if FORBIDDEN.search(cleaned):
        raise ValueError("comando proibido na consulta")
    return cleaned
```
`ailab/db.py`:
```python
import sqlite3
from pathlib import Path

SCHEMA = """
CREATE TABLE IF NOT EXISTS runs(
  id INTEGER PRIMARY KEY, ts TEXT DEFAULT CURRENT_TIMESTAMP, automation TEXT, model TEXT,
  variant TEXT, ok INTEGER, manual_seconds REAL, actual_seconds REAL);
CREATE TABLE IF NOT EXISTS bench(
  id INTEGER PRIMARY KEY, ts TEXT DEFAULT CURRENT_TIMESTAMP, task TEXT, model TEXT,
  variant TEXT, case_id INTEGER, correct INTEGER, valid INTEGER, seconds REAL);
"""


def connect(path="data/results.db"):
    Path(path).parent.mkdir(parents=True, exist_ok=True)
    conn = sqlite3.connect(path)
    conn.executescript(SCHEMA)
    return conn


def log_run(conn, automation, model, variant, ok, manual_seconds, actual_seconds):
    conn.execute(
        "INSERT INTO runs(automation,model,variant,ok,manual_seconds,actual_seconds) VALUES(?,?,?,?,?,?)",
        (automation, model, variant, int(ok), manual_seconds, actual_seconds))
    conn.commit()


def log_bench(conn, task, model, variant, case_id, correct, valid, seconds):
    conn.execute(
        "INSERT INTO bench(task,model,variant,case_id,correct,valid,seconds) VALUES(?,?,?,?,?,?,?)",
        (task, model, variant, case_id, int(correct), int(valid), seconds))
    conn.commit()
```

```python
import time
from dataclasses import dataclass
from ailab.db import log_run
from ailab.llm import generate, DEFAULT_MODEL
from ailab.prompts import load_template, render
from ailab.schemas import TicketTriage, Invoice, Summary
from ailab.structured import run_structured


@dataclass
class Automation:
    template: str
    schema: type
    manual_seconds: int


AUTOMATIONS = {
    "triage": Automation("triage", TicketTriage, 90),
    "extract": Automation("extract", Invoice, 120),
    "summarize": Automation("summarize", Summary, 300),
}


def run_batch(conn, name, texts, variant="few_shot", model=DEFAULT_MODEL, llm=generate):
    auto = AUTOMATIONS[name]
    tpl = load_template(auto.template)
    results = []
    for text in texts:
        start = time.perf_counter()
        res = run_structured(render(tpl, text, variant), auto.schema, model=model, llm=llm)
        log_run(conn, name, model, variant, res.ok, auto.manual_seconds,
                time.perf_counter() - start)
        results.append(res)
    return results
```

```python
import re
import sqlite3
import pandas as pd
from ailab.llm import generate
from ailab.sqlguard import validate_select

SCHEMA_HELP = """Tabelas SQLite:
runs(id, ts, automation, model, variant, ok, manual_seconds, actual_seconds)
bench(id, ts, task, model, variant, case_id, correct, valid, seconds)
ok/correct/valid são 0 ou 1. automation: triage, extract, summarize."""


def _extract_sql(text):
    m = re.search(r"```(?:sql)?\s*(.*?)```", text, re.S | re.I)
    return (m.group(1) if m else text).strip()


def ask(db_path, question, llm=generate):
    prompt = (f"{SCHEMA_HELP}\n\nEscreva UMA consulta SQLite SELECT que responda: {question}\n"
              "Responda apenas com o SQL.")
    sql = validate_select(_extract_sql(llm(prompt)))
    conn = sqlite3.connect(f"file:{db_path}?mode=ro", uri=True)
    try:
        return sql, pd.read_sql_query(sql, conn)
    finally:
        conn.close()
```

```python
import sqlite3
import pandas as pd
import streamlit as st
from ailab.askdata import ask
from ailab.db import connect

DB = "data/results.db"
connect(DB).close()
st.set_page_config(page_title="ai-lab", layout="wide")
st.title("ai-lab — IA aplicada, 100% local")

conn = sqlite3.connect(DB)
runs = pd.read_sql_query("select * from runs", conn)
bench = pd.read_sql_query("select * from bench", conn)

tab1, tab2, tab3 = st.tabs(["Horas economizadas", "Benchmark de prompts", "Pergunte aos dados"])

with tab1:
    if runs.empty:
        st.info("Rode `python -m ailab.cli automate triage --n 10` para gerar dados.")
    else:
        saved_h = (runs.manual_seconds.sum() - runs.actual_seconds.sum()) / 3600
        c1, c2, c3 = st.columns(3)
        c1.metric("Itens processados", len(runs))
        c2.metric("Horas economizadas", f"{saved_h:.2f} h")
        c3.metric("Taxa de sucesso", f"{runs.ok.mean():.0%}")
        by_auto = runs.groupby("automation").apply(
            lambda d: (d.manual_seconds.sum() - d.actual_seconds.sum()) / 3600, include_groups=False)
        st.bar_chart(by_auto.rename("horas economizadas"))
        st.dataframe(runs.tail(20), use_container_width=True)

with tab2:
    if bench.empty:
        st.info("Rode `python -m ailab.cli bench --n 15 --models qwen2.5:3b llama3.2:3b`.")
    else:
        agg = bench.groupby(["model", "variant"]).agg(
            acerto=("correct", "mean"), formato_valido=("valid", "mean"),
            latencia_s=("seconds", "mean")).reset_index()
        st.dataframe(agg.style.format({"acerto": "{:.0%}", "formato_valido": "{:.0%}", "latencia_s": "{:.1f}"}),
                     use_container_width=True)
        st.bar_chart(agg.pivot(index="variant", columns="model", values="acerto"))

with tab3:
    q = st.text_input("Pergunte algo sobre os dados", "Qual automação economizou mais tempo?")
    if st.button("Perguntar"):
        try:
            sql, df = ask(DB, q)
            st.code(sql, language="sql")
            st.dataframe(df)
        except Exception as e:
            st.error(f"Não consegui responder: {e}")
```
